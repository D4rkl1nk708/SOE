function AccuracyBadge({ pct }: { pct: number | null }) {
  if (pct === null) return <span className="text-[10px] italic text-muted-foreground opacity-40">—</span>;
  const color = pct >= 75 ? "var(--accent-green)" : pct >= 50 ? "var(--accent-amber)" : "var(--accent-red)";
  return (
    <span className="text-[10px] font-bold px-2 py-0.5 rounded-md border border-border bg-card shadow-sm tabular-nums"
      style={{ color }}>
      {pct}%
    </span>
  );
}
